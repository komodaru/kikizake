# kikizake suite d3d7rowspill: 24-bit spans reaching the next row.
# x86 only
case spill-rgb-5
d3d 7
d3ddev rgb 24 5 3
d3drs set 22 1
d3dclear 0xff010203
d3dtri 0xff6ba3fb 51 17 79 17 64 0 z 500
d3ddump
d3drs set 27 1
d3drs set 19 2
d3drs set 20 2
d3dtri 0xff1a2b3c 51 17 79 17 64 0
d3ddump
end
case depth-rgb-5
d3d 7
d3ddev rgb 24 5 3 z16
d3drs set 22 1
d3dclear 0xff010203 z 1000
d3dtri 0xff6ba3fb 51 17 79 17 64 0 z 500
d3ddump
d3dzdump
end
